% Ueberschriftenformate mit Praefix und Doppelpunkt.

  {\Large\bfseries}{\TBteilname~\thesection:}{0.6em}{}

  {\Large\bfseries}{\TBteilname~\thesubsection:}{0.6em}{}

% Hauptteil-Trenner: Trennerseite + part-Eintrag im IHV + Praefix-Umschaltung
% #1 Titel (z. B. "Anlagen"), #2 Praefix (z. B. "Anlage"), #3 Untertext
\newcommand{\TBhauptteil}[3]{%
  \clearpage
  \renewcommand{\TBteilname}{#2}%
  % Eigener hyperref-Ankernamensraum je Hauptteil: Abschnitte und Anlagen
  % tragen dieselben Nummern (z. B. 2.3.2), erzeugten dadurch aber gleich
  % benannte PDF-Sprungziele -- Verweise auf eine Anlage landeten beim
  % gleichnummerierten Abschnitt. Der Teil-Praefix macht die Ziele eindeutig.
  \renewcommand{\theHsection}{#2.\thesection}%
  \renewcommand{\theHsubsection}{#2.\thesubsection}%
  \renewcommand{\theHsubsubsection}{#2.\thesubsubsection}%
  \addtocontents{toc}{\protect\renewcommand{\protect\cftsecpresnum}{#2~}}%
  \addtocontents{toc}{\protect\renewcommand{\protect\cftsubsecpresnum}{#2~}}%
  \addtocontents{toc}{\protect\renewcommand{\protect\cftsubsubsecpresnum}{#2~}}%
  \phantomsection
  \addcontentsline{toc}{part}{#1}%
  \markright{#1}%
  \vspace*{0.32\textheight}
  \begin{center}{\Huge\bfseries #1}\end{center}
  \begin{center}\begin{minipage}{0.8\textwidth}\centering #3\end{minipage}\end{center}
  \clearpage
}
